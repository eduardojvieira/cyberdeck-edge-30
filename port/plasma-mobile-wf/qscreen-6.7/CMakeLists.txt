cmake_minimum_required(VERSION 3.20)
project(eqs_qscreen LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  message(FATAL_ERROR "Use the prepared ARM64 SDK, not the PC compiler")
endif()
set(EQS_QSCREEN_SOURCE_DIR "" CACHE PATH "Directory containing the verified, patched src/ files")
if(NOT EXISTS "${EQS_QSCREEN_SOURCE_DIR}/src/abstractbackend.h")
  message(FATAL_ERROR "Prepare the pinned sources first (see README.md)")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
find_package(Qt6 6.10.2 EXACT REQUIRED COMPONENTS Core Gui GuiPrivate)
find_package(KF6Screen 6.7.4 EXACT REQUIRED)
execute_process(COMMAND dpkg-query -W "-f=\${Version}" libkf6screen8
  OUTPUT_VARIABLE installed_version RESULT_VARIABLE status)
if(NOT status EQUAL 0 OR NOT installed_version STREQUAL "4:6.7.4-1")
  message(FATAL_ERROR "libkf6screen8 must be exactly 4:6.7.4-1 (private backend ABI)")
endif()
add_compile_options(-fstack-protector-strong -D_FORTIFY_SOURCE=2)
add_link_options(-Wl,-z,relro,-z,now)
add_library(KSC_QScreen MODULE ${EQS_QSCREEN_SOURCE_DIR}/src/qscreenbackend.cpp ${EQS_QSCREEN_SOURCE_DIR}/src/qscreenconfig.cpp ${EQS_QSCREEN_SOURCE_DIR}/src/qscreenscreen.cpp ${EQS_QSCREEN_SOURCE_DIR}/src/qscreenoutput.cpp)
target_include_directories(KSC_QScreen PRIVATE ${EQS_QSCREEN_SOURCE_DIR}/src /usr/include/KF6/KScreen/kscreen)
target_link_libraries(KSC_QScreen PRIVATE Qt6::Core Qt6::Gui KF6::Screen)
target_link_options(KSC_QScreen PRIVATE -Wl,--as-needed)
set_target_properties(KSC_QScreen PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/kf6/kscreen")
add_executable(test-qscreen test-qscreen.cpp)
target_include_directories(test-qscreen PRIVATE ${EQS_QSCREEN_SOURCE_DIR}/src /usr/include/KF6/KScreen/kscreen)
target_link_libraries(test-qscreen PRIVATE Qt6::Core Qt6::Gui Qt6::GuiPrivate KF6::Screen)

install(TARGETS KSC_QScreen LIBRARY DESTINATION lib/aarch64-linux-gnu/qt6/plugins/kf6/kscreen)
install(FILES copyright DESTINATION share/doc/eqs-kscreen-qscreen)
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_PACKAGE_NAME eqs-kscreen-qscreen)
set(CPACK_PACKAGE_VERSION "6.7.4-0+eqs2")
set(CPACK_PACKAGE_CONTACT "eqs port maintainers <eqs-port@localhost>")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Read-only QScreen backend for eqs Plasma on Wayfire")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE arm64)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
# AbstractBackend is a private API even though the public SONAME is unchanged.
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libkf6screen8 (= 4:6.7.4-1), libkscreen-bin (= 4:6.7.4-1)")
set(CPACK_STRIP_FILES TRUE)
include(CPack)
